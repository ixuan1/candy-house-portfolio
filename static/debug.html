④ 常用 curl 命令（服务器上排错用）

# 健康检查
curl -s http://localhost/api/health

# 列出全部作品
curl -s http://localhost/api/works

# 新增一件作品（POST）
curl -s -X POST http://localhost/api/works \
  -H 'Content-Type: application/json' \
  -d '{"title":"测试糖果","category":"Web","color":"#FF1493","description":"调试用","link":""}'

# 改（PUT，id 换成真实 id）
curl -s -X PUT http://localhost/api/works/1 \
  -H 'Content-Type: application/json' -d '{"title":"改名后的糖果"}'

# 删（DELETE）
curl -s -X DELETE http://localhost/api/works/1

⑤ 容器/服务状态（在服务器上跑）

# 看所有容器状态（STATUS 应全是 Up，app 应有 2 个）
docker compose ps

# 看日志（加 --follow 实时滚动）
docker compose logs -f app
docker compose logs -f nginx

# 探活某容器
docker compose exec app curl -s http://localhost:8000/api/health
